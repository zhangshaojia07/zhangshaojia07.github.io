{{/* Local image render hook; adds responsive images to Markdown images. */}} {{- $dest := .Destination -}} {{- $res := false -}} {{- $u := urls.Parse $dest -}} {{- if and (not $u.IsAbs) .Page -}} {{- with .Page.Resources.GetMatch $u.Path -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- if and $res (reflect.IsImageResource $res) -}} {{- $set := partial "img-set.html" (dict "image" $res) -}} {{- else -}} {{- end -}}
